--// SKYBOX
local Lighting = game:GetService("Lighting")

local skyData = {
	SkyboxBk = "rbxassetid://4498828382",
	SkyboxDn = "rbxassetid://4498828812",
	SkyboxFt = "rbxassetid://4498829917",
	SkyboxLf = "rbxassetid://4498830911",
	SkyboxRt = "rbxassetid://4498830417",
	SkyboxUp = "rbxassetid://4498831746"
}

local function applySkybox()
	local sky = Lighting:FindFirstChildOfClass("Sky")

	if not sky then
		sky = Instance.new("Sky")
		sky.Parent = Lighting
	end

	for property, value in pairs(skyData) do
		sky[property] = value
	end
end

applySkybox()

Lighting.ChildRemoved:Connect(function(child)
	if child:IsA("Sky") then
		task.wait()
		applySkybox()
	end
end)

Lighting.ChildAdded:Connect(function(child)
	if child:IsA("Sky") then
		task.wait()
		applySkybox()
	end
end)


--// ASSET REPLACEMENT

local function getAsset(url: string): string
	local success, result = pcall(function()
		if writefile and isfile and getcustomasset then
			local fileName = "tex_" .. tostring(string.len(url) % 99999) .. "_" .. tostring(string.byte(url, 10) or 1) .. ".asset"

			if not isfile(fileName) then
				local data = game:HttpGet(url)

				if data and #data > 80 then
					writefile(fileName, data)
				end
			end

			local asset = getcustomasset(fileName)

			if asset and asset ~= "" then
				return asset
			end
		end

		return url
	end)

	return (success and result) or url
end

local IdMap: {[string]: string} = {}

for _, rule in ipairs(Rules) do
	local asset = getAsset(rule.url)

	for _, id in ipairs(rule.ids) do
		IdMap[tostring(id)] = asset
	end
end

local function apply(obj: Instance)
	if not obj then
		return
	end

	local function replace(property: string)
		local success, value = pcall(function()
			return obj[property]
		end)

		if not success or type(value) ~= "string" then
			return
		end

		local id = string.match(value, "%d+")

		if id and IdMap[id] then
			pcall(function()
				obj[property] = IdMap[id]
			end)
		end
	end

	if obj:IsA("Sound") then
		replace("SoundId")

	elseif obj:IsA("ImageLabel") or obj:IsA("ImageButton") then
		replace("Image")

	elseif obj:IsA("Decal") or obj:IsA("Texture") then
		replace("Texture")

	elseif obj:IsA("MeshPart") then
		replace("TextureID")
	end
end

task.wait(2)

for _, v in ipairs(game:GetDescendants()) do
	task.spawn(apply, v)
end

game.DescendantAdded:Connect(function(obj)
	task.defer(apply, obj)
end)
